class Solution:
    def mySqrt(self, x: int) -> int:
        if x < 2:
            return x
            
        # Integer division
        left, right = 0, x // 2

        # Implementation of Binary search
        while left <= right:
            mid = (left + right) // 2
            # Return if found
            if mid * mid == x:
                return mid
            # Case for larger values
            elif mid * mid < x:
                left = mid + 1
            # Case for smaller values
            else:
                right = mid - 1

        return right
