# Definition for singly-linked list.
# class ListNode:
#     def __init__(self, val=0, next=None):
#         self.val = val
#         self.next = next
class Solution:
    def addTwoNumbers(self, l1: Optional[ListNode], l2: Optional[ListNode]) -> Optional[ListNode]:
        # newL for new list
        newL = ListNode(0)
        # Value pointer for creating the newL
        current = newL
        # Variable to track a carry value if needed
        addOn = 0
        
        # Iterate as long as there are nodes in l1 or l2, or a addOn remains
        while l1 or l2 or addOn:
            # Get the current values; default to 0 for when list is empty
            val1 = l1.val if l1 else 0
            val2 = l2.val if l2 else 0
            
            # Calculate the sum and the new addOn
            total = val1 + val2 + addOn
            addOn = total // 10
            new_digit = total % 10
            
            # Add the new digit to the result list
            current.next = ListNode(new_digit)
            current = current.next
            
            # Move to the next nodes in the input lists
            if l1:
                l1 = l1.next
            if l2:
                l2 = l2.next
        
        # Return the newList at the next node
        return newL.next
