class Solution:
    def numIslands(self, grid: List[List[str]]) -> int:
        if not grid:
            return 0
        
        #-----------------------------------------------------
        # dfs function
        def dfs(grid, i, j):
            # check if outside of grid or at waterline
            if i < 0 or i >= len(grid) or j < 0 or j >= len(grid[0]) or grid[i][j] == '0':
                return
            
            # mark cell visited
            grid[i][j] = '0'
            
            # go every direction
            dfs(grid, i + 1, j)  
            dfs(grid, i - 1, j)  
            dfs(grid, i, j + 1)  
            dfs(grid, i, j - 1)  
        #-----------------------------------------------------
        
        num_islands = 0
        
        for i in range(len(grid)):
            for j in range(len(grid[0])):
                # get land cell
                if grid[i][j] == '1':
                    num_islands += 1
                    # find other land cells near
                    dfs(grid, i, j)
        
        return num_islands
