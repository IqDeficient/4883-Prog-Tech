class Solution:
    def outerTrees(self, trees: List[List[int]]) -> List[List[int]]:
        # edge case handling
        if len(trees) <= 1:
            return trees

        # Sort points by x then y coordinates
        trees.sort()

        # function to get orientation
        def orientation(p, q, r):
            return (q[1] - p[1]) * (r[0] - q[0]) - (q[0] - p[0]) * (r[1] - q[1])

        # Build the lower hull
        lower = []
        for point in trees:
            while len(lower) >= 2 and orientation(lower[-2], lower[-1], point) < 0:
                lower.pop()
            lower.append(point)

        # Build the upper hull
        upper = []
        for point in reversed(trees):
            while len(upper) >= 2 and orientation(upper[-2], upper[-1], point) < 0:
                upper.pop()
            upper.append(point)

        # Combine lower and upper hulls while removing duplicates
        result = lower[:-1] + upper[:-1]
        return [list(tup) for tup in sorted(set(map(tuple, result)))]
